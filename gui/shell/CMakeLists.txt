# ------------------------------------------------------------------------------------------------- WebView2 SDK
# The Microsoft.Web.WebView2 NuGet package gives WebView2.h, WebView2LoaderStatic.lib (MSVC) and WebView2Loader.dll.
# MSVC links the static loader (no extra DLL to ship); MinGW cannot link that MSVC library, so it loads
# WebView2Loader.dll at run time (LoadLibraryW + GetProcAddress) and the DLL ships next to the exe.
# Without the package (offline configure) the checked-in webview2_subset.h is compiled instead (same declarations,
# generated by tools/webview2_subset.py) and the dynamic loader is used.
set(VP_WEBVIEW2_VERSION "1.0.2903.40" CACHE STRING "Microsoft.Web.WebView2 NuGet version")
set(VP_WEBVIEW2_SDK_DIR "" CACHE PATH "Unpacked Microsoft.Web.WebView2 package (skips the download)")
option(VP_WEBVIEW2_DOWNLOAD "Download the WebView2 SDK from nuget.org at configure time" ON)

set(_wv2_dir "${VP_WEBVIEW2_SDK_DIR}")
if(NOT _wv2_dir)
  set(_wv2_dir "${CMAKE_BINARY_DIR}/_deps/webview2-${VP_WEBVIEW2_VERSION}")
  if(VP_WEBVIEW2_DOWNLOAD AND NOT EXISTS "${_wv2_dir}/build/native/include/WebView2.h")
    set(_wv2_pkg "${CMAKE_BINARY_DIR}/_deps/webview2-${VP_WEBVIEW2_VERSION}.zip")
    set(_wv2_url "https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/${VP_WEBVIEW2_VERSION}")
    message(STATUS "Downloading WebView2 SDK ${VP_WEBVIEW2_VERSION} from nuget.org")
    file(DOWNLOAD "${_wv2_url}" "${_wv2_pkg}" STATUS _wv2_status TIMEOUT 180 INACTIVITY_TIMEOUT 30 TLS_VERIFY ON)
    list(GET _wv2_status 0 _wv2_code)
    if(_wv2_code EQUAL 0)
      file(ARCHIVE_EXTRACT INPUT "${_wv2_pkg}" DESTINATION "${_wv2_dir}"
           PATTERNS "build/native/include/*" "build/native/x64/*" "LICENSE.txt")
    else()
      list(GET _wv2_status 1 _wv2_msg)
      message(WARNING "WebView2 SDK download failed (${_wv2_msg}); using gui/shell/webview2_subset.h")
    endif()
    file(REMOVE "${_wv2_pkg}")
  endif()
endif()

set(VP_WEBVIEW2_LOADER_DLL "")
set(VP_WEBVIEW2_LICENSE "")
set(VP_WEBVIEW2_STATIC 0)
if(EXISTS "${_wv2_dir}/build/native/include/WebView2.h")
  set(VP_WEBVIEW2_HAVE_SDK 1)
  set(VP_WEBVIEW2_INCLUDE "${_wv2_dir}/build/native/include")
  if(EXISTS "${_wv2_dir}/LICENSE.txt")
    set(VP_WEBVIEW2_LICENSE "${_wv2_dir}/LICENSE.txt")
  endif()
  if(MSVC AND EXISTS "${_wv2_dir}/build/native/x64/WebView2LoaderStatic.lib")
    set(VP_WEBVIEW2_STATIC 1)
    set(VP_WEBVIEW2_STATIC_LIB "${_wv2_dir}/build/native/x64/WebView2LoaderStatic.lib")
  elseif(EXISTS "${_wv2_dir}/build/native/x64/WebView2Loader.dll")
    set(VP_WEBVIEW2_LOADER_DLL "${_wv2_dir}/build/native/x64/WebView2Loader.dll")
  endif()
  if(VP_WEBVIEW2_STATIC)
    message(STATUS "WebView2 SDK: ${_wv2_dir} (static loader WebView2LoaderStatic.lib)")
  else()
    message(STATUS "WebView2 SDK: ${_wv2_dir} (dynamic loader WebView2Loader.dll next to the exe)")
  endif()
else()
  set(VP_WEBVIEW2_HAVE_SDK 0)
  message(STATUS "WebView2 SDK not available: compiling against gui/shell/webview2_subset.h; "
                 "put an x64 WebView2Loader.dll next to VetusPoeta.exe yourself")
endif()

# ------------------------------------------------------------------------------------------------- executable
add_executable(VetusPoeta WIN32
  app.cpp
  engine_host.cpp
  shell_util.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/VetusPoeta.rc")
if(MSVC)
  # A .manifest source is passed to the linker as /MANIFESTINPUT and merged into the embedded manifest
  # (per-monitor-v2 DPI awareness, common controls v6, UTF-8 code page, long paths).
  target_sources(VetusPoeta PRIVATE "${VP_MANIFEST_PATH}")
  target_link_options(VetusPoeta PRIVATE /MANIFESTUAC:NO)
else()
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/VetusPoeta.rc" APPEND PROPERTY COMPILE_DEFINITIONS VP_RC_EMBED_MANIFEST)
endif()
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/VetusPoeta.rc" PROPERTIES
  OBJECT_DEPENDS "${VP_ICON_PATH};${VP_MANIFEST_PATH}")
target_include_directories(VetusPoeta PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(VetusPoeta PRIVATE
  UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00
  VP_WEBVIEW2_HAVE_SDK=${VP_WEBVIEW2_HAVE_SDK} VP_WEBVIEW2_STATIC=${VP_WEBVIEW2_STATIC}
  VP_SHELL_VERSION="${PROJECT_VERSION}")
if(VP_WEBVIEW2_HAVE_SDK)
  target_include_directories(VetusPoeta SYSTEM PRIVATE "${VP_WEBVIEW2_INCLUDE}")
  if(NOT CMAKE_HOST_WIN32)
    # WebView2.h includes "EventToken.h"; MinGW ships it as eventtoken.h (case matters on Linux).
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wv2shim/EventToken.h" "#include <eventtoken.h>\n")
    target_include_directories(VetusPoeta SYSTEM PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/wv2shim")
  endif()
endif()
if(VP_WEBVIEW2_STATIC)
  target_link_libraries(VetusPoeta PRIVATE "${VP_WEBVIEW2_STATIC_LIB}" version)
endif()
target_link_libraries(VetusPoeta PRIVATE vp_host_logic
  ole32 oleaut32 uuid shell32 shlwapi dwmapi user32 gdi32 advapi32 comdlg32)
if(NOT MSVC)
  target_compile_options(VetusPoeta PRIVATE -Wall -Wextra -Wshadow)
  target_link_options(VetusPoeta PRIVATE -municode)
endif()
if(TARGET vpengine)
  add_dependencies(VetusPoeta vpengine)
endif()

# ------------------------------------------------- runnable folder in the build tree: <build>/app[/<config>]
# VetusPoeta.exe, vpengine.exe, WebView2Loader.dll (MinGW) and ui/ side by side (the dist layout without data).
find_package(Python3 COMPONENTS Interpreter)
get_property(_multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(_multi)
  set(VP_APP_DIR "${CMAKE_BINARY_DIR}/app/$<CONFIG>")
else()
  set(VP_APP_DIR "${CMAKE_BINARY_DIR}/app")
endif()
set(_app_cmds
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${VP_APP_DIR}"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:VetusPoeta>" "${VP_APP_DIR}")
if(TARGET vpengine)
  list(APPEND _app_cmds COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:vpengine>" "${VP_APP_DIR}")
endif()
if(VP_WEBVIEW2_LOADER_DLL)
  list(APPEND _app_cmds COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${VP_WEBVIEW2_LOADER_DLL}" "${VP_APP_DIR}")
endif()
if(Python3_FOUND)
  list(APPEND _app_cmds COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/pack_ui.py" --out "${VP_APP_DIR}/ui" --quiet)
else()
  message(WARNING "Python 3 not found: <build>/app/ui is not assembled (run tools/pack_ui.py by hand)")
endif()
add_custom_target(vp_app ALL ${_app_cmds}
  COMMENT "Assembling ${VP_APP_DIR} (VetusPoeta.exe, vpengine.exe, ui/)"
  VERBATIM)
add_dependencies(vp_app VetusPoeta)
if(TARGET vpengine)
  add_dependencies(vp_app vpengine)
endif()
