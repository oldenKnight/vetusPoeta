cmake_minimum_required(VERSION 3.20)
project(VetusPoeta VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(VP_BUILD_GUI   "Build the Windows GUI (WebView2 shell)" ${WIN32})
option(VP_BUILD_TESTS "Build unit tests" ON)
option(VP_SANITIZE    "Build with ASan+UBSan (+LSan)" OFF)
option(VP_WITH_LLM    "Build the local-model engine (vendored llama.cpp)" ON)

if(MSVC)
  add_compile_options(/W4 /utf-8 /EHsc /MP)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  add_compile_options(-Wall -Wextra -Wshadow)
  if(VP_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
    add_link_options(-fsanitize=address,undefined)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Windows")  # MinGW cross build
    add_link_options(-static -static-libgcc -static-libstdc++)
  endif()
endif()

find_package(Threads REQUIRED)
add_library(vp_third_party INTERFACE)
target_include_directories(vp_third_party INTERFACE ${CMAKE_SOURCE_DIR}/third_party)

# Modules are added when their directory has a CMakeLists.txt, so the tree builds while modules are still arriving.
foreach(mod core lex subs nlp rules llm online cli)
  if(EXISTS ${CMAKE_SOURCE_DIR}/engine/${mod}/CMakeLists.txt)
    if(mod STREQUAL "llm" AND NOT VP_WITH_LLM)
      continue()
    endif()
    add_subdirectory(engine/${mod})
  endif()
endforeach()
if(VP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(engine/tests)
endif()
if(EXISTS ${CMAKE_SOURCE_DIR}/gui/shell/CMakeLists.txt)
  add_subdirectory(gui/shell)
endif()
